  ```lean
  def SlopePred (t ℓ : ℚ) (c : Fin 4 → ℤ) : Prop :=
    t ∉ contactSet ∧ t ≠ 1 ∧ ℓ ≠ 0 ∧ ∃ (η q : ℚ) (hz : Fin 6 → ℤ × (Fin 3 → ℤ)),
      (η = 1 ∨ η = -1) ∧
      let R0 := η * contactPoly t / (t - 1) ^ 8; let a := 1 - R0 * q ^ 2; let R := ℓ^2 * a^2 * R0
      a ≠ 0 ∧ (∀ g, (hz g).1 ≠ 0 ∧ (hz g).1 ≠ c 0) ∧
      let C g j := testValues (hz g).1 (hz g).2 c j / ((R0 : K0) ^ prec j * ((t - 1 : ℚ) : K0) ^ (prec j + 1))
      let W g := ∏ j, (1 + (R : K0) * C g j ^ 2)
      let Θ : ℚ := 2 * (∑ g, (W g)⁻¹).re
      (∀ g, W g ≠ 0) ∧ Θ ≠ 0 ∧ PoleSoluble (R / Θ ^ 2)
  ```
