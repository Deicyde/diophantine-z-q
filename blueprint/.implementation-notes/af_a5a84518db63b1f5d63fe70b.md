  theorem masked_register_iff (c q R : ℕ) (hc : 0 < c) :
      let K := 2^c; let B := 4*K; let F := ∑ t in range (q+1), B^t; let D := (K-1)*F
      (Nat.land R D = R ∧ R < B^q) ↔ ∃ ρ : ℕ → ℕ, (∀ t ≤ q, ρ t < K) ∧ ρ q = 0 ∧
        R = ∑ t in range (q+1), ρ t * B^t
  theorem F_eq_geom (B F T q : ℕ) (hB : 2 ≤ B) (hT : T = B^q) (h : B*F + 1 = F + B*T) :
      F = ∑ t in range (q+1), B^t
