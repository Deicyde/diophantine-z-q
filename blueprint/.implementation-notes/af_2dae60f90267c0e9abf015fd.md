  theorem kappa_eval_eq_zero_iff (φ : Formula) (x : ℕ → ℤ) :
      MvPolynomial.eval x (kappa φ) = 0 ↔ φ.Holds x
  theorem exists_block_iff (prog : NatM Formula) (s : BState) (hargs : ArgsBelow prog s.next) :
      ∀ x : ℕ → ℕ, (∃ y : ℕ → ℕ, (prog.run' s).Holds (override x s.next (prog.run s).2.next y))
        ↔ (prog.rel) x   -- stated per connective, see the article

  (ℕ → ℕ) → Prop` defined by structural recursion that quantifies the variables of each
  construction step inside its conjunct, with the theorem
  `(∃ y, φ.Holds (x[y])) ↔ φ.HoldsSome x` under the scope hypotheses (fresh intervals,
  no use outside).
