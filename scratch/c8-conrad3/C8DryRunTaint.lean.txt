  match ci with

partial def stripLam : Expr → Expr
  | .lam _ _ b _ => stripLam b
  | .mdata _ b => stripLam b
  | e => e

def projField (env : Environment) (n : Name) : Option String :=
  match env.getProjectionFnInfo? n, n with
  | some _, .str _ s => some s
  | _, _ => none
